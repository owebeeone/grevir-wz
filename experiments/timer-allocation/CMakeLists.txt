cmake_minimum_required(VERSION 3.20)
project(grevir_timer_prototype LANGUAGES CXX)
enable_testing()
add_executable(timer_prototype_checks static_checks.cpp oracle_checks.cpp)
add_executable(timer_frequency_checks frequency_checks.cpp)
add_executable(timer_atmega328p_checks atmega328p_checks.cpp)
target_compile_definitions(timer_atmega328p_checks PRIVATE HAS_STD_LIB=1)
# Use the live workspace headers so these checks exercise the real driver.
foreach(member base time core peripherals registers avr)
  target_include_directories(timer_atmega328p_checks SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../grevir-${member}/src")
endforeach()
target_include_directories(timer_atmega328p_checks PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../grevir-avr/tests" "${CMAKE_CURRENT_SOURCE_DIR}/../../grevir-test-support/include")
option(TIMER_PROTOTYPE_SANITIZERS "Enable native address/undefined sanitizers" OFF)
foreach(target timer_prototype_checks timer_frequency_checks timer_atmega328p_checks)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../grevir-peripherals/src" "${CMAKE_CURRENT_SOURCE_DIR}/../../grevir-core/src")
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(TIMER_PROTOTYPE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME timer_atmega328p_binding COMMAND timer_atmega328p_checks)
add_test(NAME timer_prototype_oracle COMMAND timer_prototype_checks)
add_test(NAME timer_frequency_intersection COMMAND timer_frequency_checks)
add_test(NAME timer_prototype_rejections COMMAND "${CMAKE_COMMAND}"
  "-DCOMPILER=${CMAKE_CXX_COMPILER}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
  "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/check_rejections.cmake")
