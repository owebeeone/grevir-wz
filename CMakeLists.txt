cmake_minimum_required(VERSION 3.20)
project(grevir_workspace LANGUAGES CXX)

# Only extracted libraries participate at this stage.
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "avr")
  option(GREVIR_BUILD_COMPILE_CHECKS "Compile native headers and template users" OFF)
  option(GREVIR_BUILD_HOST_TESTS "Run deterministic native behavior tests" OFF)
else()
  option(GREVIR_BUILD_COMPILE_CHECKS "Compile native headers and template users" ON)
  option(GREVIR_BUILD_HOST_TESTS "Run deterministic native behavior tests" OFF)
endif()
if(GREVIR_BUILD_HOST_TESTS)
  enable_testing()
  add_subdirectory(grevir-test-support)
endif()
add_subdirectory(grevir-base)
add_subdirectory(grevir-time)
add_subdirectory(grevir-core)
add_subdirectory(grevir-peripherals)
add_subdirectory(grevir-registers)
add_subdirectory(grevir-avr)

add_subdirectory(grevir-pulse-codec)
add_subdirectory(grevir-pulse-io)
add_subdirectory(grevir-packet)
add_subdirectory(grevir-encoder)
add_subdirectory(grevir-stepper)
add_subdirectory(grevir-arduino)
add_subdirectory(grevir-arduino-avr)
add_subdirectory(grevir-fastled)

if(CMAKE_SYSTEM_PROCESSOR STREQUAL "avr")
  add_subdirectory(avr-probe)
endif()
